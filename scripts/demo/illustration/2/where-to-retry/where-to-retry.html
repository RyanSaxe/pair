A timeout leaves the charge unknown, because the gateway may have charged the card. chargeOrder retries only a GatewayTimeout.

Where should the retry run?

export async function chargeOrder(order: Order) {
  const idempotencyKey = `order-${order.id}`;
  for (let attempt = 1; ; attempt++) {
    try {
      return await gateway.charge(order, { idempotencyKey });
    } catch (err) {
      if (!isTimeout(err) || attempt === 3) throw err;
      await sleep(200 * 2 ** attempt);
    }
  }
}
